cmake_minimum_required(VERSION 3.16)
project(spelling_blizzard_recomp C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(SDL2 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FREETYPE REQUIRED IMPORTED_TARGET freetype2)
find_package(Threads REQUIRED)

# sb_embed(OUT_C SYMBOL FILE...): the files' bytes, one after another, as a
# NUL-terminated `const char SYMBOL[]` in OUT_C. Changing a file configures again.
function(sb_embed out_c symbol)
    set(hex "")
    foreach(f ${ARGN})
        file(READ ${f} h HEX)
        string(APPEND hex "${h}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${f})
    endforeach()
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
    file(WRITE ${out_c}.new "/* Generated by CMake (sb_embed). */\nconst char ${symbol}[] = {${hex}0};\n")
    configure_file(${out_c}.new ${out_c} COPYONLY)
endfunction()

# ---------------------------------------------------------------------------
# Host runtime: CPU helpers, loader and the Win16 API layer. It contains no
# game code and builds without the game files.
# ---------------------------------------------------------------------------
# (Object files, not an archive: the weak API stubs must never win over the
# real implementations.)
add_library(sb_recomp_runtime OBJECT
    src/recomp/main.c src/recomp/runtime.c src/recomp/loader.c src/recomp/kernel.c
    src/recomp/hostfs.c src/recomp/user.c src/recomp/gdi.c src/recomp/mmsystem.c src/recomp/api_stubs.c)
target_include_directories(sb_recomp_runtime PUBLIC src/recomp src/recomp/imports)
target_link_libraries(sb_recomp_runtime PUBLIC SDL2::SDL2 PkgConfig::FREETYPE)
target_compile_options(sb_recomp_runtime PRIVATE -Wall -Wno-unused-function)

# ---------------------------------------------------------------------------
# The recompiler (src/recomp/gen): BLIZZARD.EXE -> C. No game code in it either.
# ---------------------------------------------------------------------------
set(SB_GEN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen)
set(SB_GEN_SOURCES ${SB_GEN_DIR}/gen.c ${SB_GEN_DIR}/ne.c ${SB_GEN_DIR}/x86.c ${SB_GEN_DIR}/exports.c)
set(SB_GEN_FUNCTIONS_C ${CMAKE_CURRENT_BINARY_DIR}/gen_functions_txt.c)
sb_embed(${SB_GEN_FUNCTIONS_C} gen_functions_txt ${SB_GEN_DIR}/functions.txt)

add_library(sb_gen_lib STATIC ${SB_GEN_SOURCES} ${SB_GEN_FUNCTIONS_C})
target_include_directories(sb_gen_lib PUBLIC ${SB_GEN_DIR})
target_compile_options(sb_gen_lib PRIVATE -Wall -Wextra)
add_executable(sb_gen ${SB_GEN_DIR}/sb_gen.c)
target_link_libraries(sb_gen PRIVATE sb_gen_lib)
target_compile_options(sb_gen PRIVATE -Wall -Wextra)

# ---------------------------------------------------------------------------
# TinyCC (third_party/tinycc, LGPL-2.1), as libtcc: compiles the recompiled
# code in memory when the game starts.
# ---------------------------------------------------------------------------
set(TCC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/tinycc)
if(NOT EXISTS ${TCC_DIR}/libtcc.c)
    message(FATAL_ERROR "third_party/tinycc is empty: run `git submodule update --init`.")
endif()
set(TCC_GEN ${CMAKE_CURRENT_BINARY_DIR}/tinycc)
file(STRINGS ${TCC_DIR}/VERSION TCC_VERSION LIMIT_COUNT 1)
# What TinyCC's configure would write. The target is the host's (tcc.h picks it).
file(WRITE ${TCC_GEN}/config.h.new
     "#define TCC_VERSION \"${TCC_VERSION}\"\n"
     "#define CONFIG_TCC_PREDEFS 1\n"    # predefined macros built in: no files at run time
     "#define CONFIG_TCC_BACKTRACE 0\n"
     "#define CONFIG_TCC_BCHECK 0\n"
     "#define CONFIG_TCCDIR \"\"\n")
configure_file(${TCC_GEN}/config.h.new ${TCC_GEN}/config.h COPYONLY)
# tccdefs_.h: include/tccdefs.h as C strings, made by TinyCC's c2str.
try_compile(TCC_C2STR_BUILT ${TCC_GEN}/c2str_build
    SOURCES ${TCC_DIR}/conftest.c
    COMPILE_DEFINITIONS -DC2STR
    OUTPUT_VARIABLE TCC_C2STR_LOG
    COPY_FILE ${TCC_GEN}/c2str)
if(NOT TCC_C2STR_BUILT)
    message(FATAL_ERROR "Building TinyCC's c2str failed:\n${TCC_C2STR_LOG}")
endif()
execute_process(COMMAND ${TCC_GEN}/c2str ${TCC_DIR}/include/tccdefs.h ${TCC_GEN}/tccdefs_.h
                RESULT_VARIABLE TCC_C2STR_RESULT)
if(NOT TCC_C2STR_RESULT EQUAL 0)
    message(FATAL_ERROR "TinyCC's c2str failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${TCC_DIR}/include/tccdefs.h ${TCC_DIR}/VERSION)

add_library(tcc STATIC ${TCC_DIR}/libtcc.c)
target_include_directories(tcc PRIVATE ${TCC_GEN} INTERFACE ${TCC_DIR})
target_compile_definitions(tcc PRIVATE ONE_SOURCE=1)
target_compile_options(tcc PRIVATE -w -fno-strict-aliasing)
target_link_libraries(tcc PRIVATE ${CMAKE_DL_LIBS} Threads::Threads m)

# ---------------------------------------------------------------------------
# sb_recomp: the game, recompiled from your BLIZZARD.EXE when it starts and
# compiled in memory. This program contains no game code; it builds without
# the game files and is what releases ship.
# ---------------------------------------------------------------------------
set(SB_JIT_PRELUDE_C ${CMAKE_CURRENT_BINARY_DIR}/jit_prelude.c)
sb_embed(${SB_JIT_PRELUDE_C} sb_jit_prelude
         src/recomp/jit_prelude.h src/recomp/cpu.h src/recomp/imports/api_decls.h)
add_executable(sb_recomp $<TARGET_OBJECTS:sb_recomp_runtime> src/recomp/jit.c ${SB_JIT_PRELUDE_C})
target_include_directories(sb_recomp PRIVATE src/recomp src/recomp/imports)
target_compile_options(sb_recomp PRIVATE -Wall)
target_link_libraries(sb_recomp PRIVATE sb_gen_lib tcc SDL2::SDL2 PkgConfig::FREETYPE m)

set(SB_GAME_DIR ${CMAKE_CURRENT_SOURCE_DIR}/original CACHE PATH "Folder with your BLIZZARD.EXE and SB_CDRES.DLL")
if(NOT EXISTS ${SB_GAME_DIR}/BLIZZARD.EXE)
    message(STATUS "${SB_GAME_DIR}/BLIZZARD.EXE not found: no tests, and no sb_recomp_aot. "
                   "sb_recomp itself doesn't need it to build.")
    return()
endif()

# ---------------------------------------------------------------------------
# sb_recomp_aot (development): the same, but the recompiled C is generated
# into the build tree and compiled ahead of time with the host compiler, so a
# debugger sees every f_SSSS_OOOO. Never distributed. The generated file names
# depend on the EXE, so the recompiler is built and run while configuring;
# changing the EXE or the recompiler configures again.
# ---------------------------------------------------------------------------
set(RECOMP_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/recomp_gen)
try_compile(SB_GEN_BUILT ${CMAKE_CURRENT_BINARY_DIR}/sb_gen_configure
    SOURCES ${SB_GEN_SOURCES} ${SB_GEN_DIR}/sb_gen.c ${SB_GEN_FUNCTIONS_C}
    COMPILE_DEFINITIONS -O2
    C_STANDARD 11
    OUTPUT_VARIABLE SB_GEN_LOG
    COPY_FILE ${CMAKE_CURRENT_BINARY_DIR}/sb_gen_configure/sb_gen)
if(NOT SB_GEN_BUILT)
    message(FATAL_ERROR "Building the recompiler failed:\n${SB_GEN_LOG}")
endif()
execute_process(COMMAND ${CMAKE_CURRENT_BINARY_DIR}/sb_gen_configure/sb_gen ${RECOMP_GEN_DIR}
                        --exe ${SB_GAME_DIR}/BLIZZARD.EXE --imports ${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/imports
                RESULT_VARIABLE RECOMP_GEN_RESULT)
if(NOT RECOMP_GEN_RESULT EQUAL 0)
    message(FATAL_ERROR "Recompiling ${SB_GAME_DIR}/BLIZZARD.EXE failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
             ${SB_GEN_SOURCES} ${SB_GEN_DIR}/sb_gen.c ${SB_GEN_DIR}/gen.h ${SB_GEN_DIR}/ne.h ${SB_GEN_DIR}/x86.h
             ${SB_GEN_DIR}/exports.h ${SB_GAME_DIR}/BLIZZARD.EXE)
file(GLOB RECOMP_GEN_SOURCES ${RECOMP_GEN_DIR}/gen_*.c)
add_library(sb_recomp_gen STATIC ${RECOMP_GEN_SOURCES})
target_include_directories(sb_recomp_gen PRIVATE ${RECOMP_GEN_DIR} src/recomp src/recomp/imports)
target_compile_options(sb_recomp_gen PRIVATE -O1 -w)

add_executable(sb_recomp_aot $<TARGET_OBJECTS:sb_recomp_runtime> src/recomp/aot.c)
target_include_directories(sb_recomp_aot PRIVATE src/recomp src/recomp/imports)
target_link_libraries(sb_recomp_aot PRIVATE sb_recomp_gen SDL2::SDL2 PkgConfig::FREETYPE m)

# Both play the opening of the game headless.
enable_testing()
foreach(prog sb_recomp sb_recomp_aot)
    add_test(NAME ${prog}_new_game_headless
             COMMAND ${prog} --headless --no-audio --game-dir ${SB_GAME_DIR}
                     --data-dir ${CMAKE_CURRENT_BINARY_DIR}/${prog}_test_run
                     --script ${CMAKE_CURRENT_SOURCE_DIR}/tests/recomp/new_game.script
             WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(${prog}_new_game_headless PROPERTIES TIMEOUT 120)
endforeach()
